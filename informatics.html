Stive și Cozi


#include <stack>
#include <queue>
#include <iostream>
using namespace std;

int main() {
    // Stivă - Last In First Out (LIFO)
    stack<int> s;
    s.push(1);
    s.push(2);
    s.push(3);
    cout << "Element de vârf: " << s.top() << endl; // 3
    s.pop();
    cout << "Noul vârf: " << s.top() << endl; // 2
    
    // Coadă - First In First Out (FIFO)
    queue<int> q;
    q.push(1);
    q.push(2);
    q.push(3);
    cout << "Primul element: " << q.front() << endl; // 1
    q.pop();
    cout << "Noul prim element: " << q.front() << endl; // 2
    
    return 0;
}

Algoritmi de Sortare


// Bubble Sort
void bubbleSort(int arr[], int n) {
    for(int i = 0; i < n-1; i++) {
        for(int j = 0; j < n-i-1; j++) {
            if(arr[j] > arr[j+1]) {
                // Interschimbare
                int temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            }
        }
    }
}

// Utilizarea funcției sort din STL

Algoritmi de Căutare


// Căutare Liniară
int cautareLiniara(int arr[], int n, int x) {
    for(int i = 0; i < n; i++) {
        if(arr[i] == x) return i;
    }
    return -1; // Nu a fost găsit
}

// Căutare Binară (vectorul trebuie să fie sortat)
int cautareBinara(int arr[], int l, int r, int x) {
    while(l <= r) {
        int mid = l + (r - l) / 2;
        
        if(arr[mid] == x) return mid;
        if(arr[mid] < x) l = mid + 1;
        else r = mid - 1;
    }
    return -1; // Nu a fost găsit
}

Algoritmi Elementari

Cel mai mare divizor comun (CMMDC)


// Metoda lui Euclid prin scăderi repetate
int cmmdc_scaderi(int a, int b) {
    while(a != b) {
        if(a > b) a -= b;
        else b -= a;
    }
    return a;
}

// Metoda lui Euclid prin împărțiri repetate
int cmmdc_impartiri(int a, int b) {
    while(b) {
        int r = a % b;
        a = b;
        b = r;
    }
    return a;
}

Cel mai mic multiplu comun (CMMMC)


int cmmmc(int a, int b) {
    return (a * b) / cmmdc_impartiri(a, b);
}

Verificare număr prim


bool estePrim(int n) {
    if(n <= 1) return false;
    if(n <= 3) return true;
    
    // Verificăm dacă n este divizibil cu 2 sau 3
    if(n % 2 == 0 || n % 3 == 0) return false;
    
    // Verificăm toți divizorii de forma 6k ± 1
    for(int i = 5; i * i <= n; i += 6) {
        if(n % i == 0 || n % (i + 2) == 0)
            return false;
    }
    
    return true;
}

Ciurul lui Eratostene


void ciurulEratostene(int n) {
    // Creăm un vector de booleeni și îl inițializăm cu true
    bool prime[n+1];
    memset(prime, true, sizeof(prime));
    
    for(int p = 2; p * p <= n; p++) {
        // Dacă prime[p] nu a fost schimbat, atunci este prim
        if(prime[p] == true) {
            // Actualizăm toți multiplii lui p
            for(int i = p * p; i <= n; i += p)
                prime[i] = false;
        }
    }
    
    // Afișăm toate numerele prime
    for(int p = 2; p <= n; p++)
        if(prime[p])
            cout << p << " ";
}

Factorizare în numere prime


void factorizarePrime(int n) {
    // Afișăm toți factorii 2
    while(n % 2 == 0) {
        cout << 2 << " ";
        n = n/2;
    }
    
    // n trebuie să fie impar acum
    for(int i = 3; i * i <= n; i += 2) {
        // Cât timp i divide n, afișăm i și împărțim n
        while(n % i == 0) {
            cout << i << " ";
            n = n/i;
        }
    }
    
    // Dacă n este un număr prim mai mare decât 2
    if(n > 2)
        cout << n << " ";
}

Calculul sumei cifrelor unui număr


int sumaCifre(int n) {
    int suma = 0;
    while(n) {
        suma += n % 10;
        n /= 10;
    }
    return suma;
}

Inversarea unui număr


int inversareNumar(int n) {
    int invers = 0;
    while(n) {
        invers = invers * 10 + n % 10;
        n /= 10;
    }
    return invers;
}

Calculul factorialului


// Metoda iterativă
int factorial(int n) {
    int rezultat = 1;
    for(int i = 2; i <= n; i++)
        rezultat *= i;
    return rezultat;
}

// Metoda recursivă
int factorialRecursiv(int n) {
    if(n <= 1) return 1;
    return n * factorialRecursiv(n-1);
}

Șirul lui Fibonacci


// Metoda iterativă
int fibonacci(int n) {
    if(n <= 1) return n;
    
    int a = 0, b = 1, c;
    for(int i = 2; i <= n; i++) {
        c = a + b;
        a = b;
        b = c;
    }
    return b;
}

// Metoda recursivă
int fibonacciRecursiv(int n) {
    if(n <= 1) return n;
    return fibonacciRecursiv(n-1) + fibonacciRecursiv(n-2);
}

Programare Orientată pe Obiecte

Clase și Obiecte


class Student {
private:
    string nume;
    int varsta;
    double nota;
    
public:
    // Constructor
    Student(string n, int v, double nt) {
        nume = n;
        varsta = v;
        nota = nt;
    }
    
    // Metode
    void afisareInfo() {
        cout << "Nume: " << nume << endl;
        cout << "Vârstă: " << varsta << endl;
        cout << "Notă: " << nota << endl;
    }
    
    // Getteri și Setteri
    string getNume() { return nume; }
    void setNume(string n) { nume = n; }
};

int main() {
    Student s1("Ion", 18, 9.5);
    s1.afisareInfo();
    return 0;
}

Moștenire


class Persoana {
protected:
    string nume;
    int varsta;
    
public:
    Persoana(string n, int v) : nume(n), varsta(v) {}
    void afisare() {
        cout << "Nume: " << nume << ", Vârstă: " << varsta << endl;
    }
};

class Student : public Persoana {
private:
    double nota;
    
public:
    Student(string n, int v, double nt) : Persoana(n, v), nota(nt) {}
    void afisare() {
        Persoana::afisare();
        cout << "Notă: " << nota << endl;
    }
};

}
